sudo nano /etc/nginx/sites-available/quantumguide

server {
    listen 80;
    listen [::]:80;
    server_name _;

    # Tambahkan batas ukuran upload (misal 500 MB)
    client_max_body_size 500M;

    client_max_body_size 510M;
    proxy_read_timeout 300s;
    proxy_connect_timeout 300s;
    proxy_send_timeout 300s;
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # Standard Proxy & WebSocket Headers
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

sudo ln -s /etc/nginx/sites-available/quantumguide /etc/nginx/sites-enabled/

sudo nginx -t

sudo systemctl reload nginx
